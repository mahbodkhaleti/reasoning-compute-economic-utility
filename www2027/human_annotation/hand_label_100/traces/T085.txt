from typing import List

def calculate_average_temperature(temperatures: List[float]) -> float:
    """
    Calculate the average temperature from a list of temperature readings.
    
    Args:
    temperatures (List[float]): A list of temperature readings.
    
    Returns:
    float: The average temperature.
    """
    if not temperatures:
        return 0.0
    
    total_sum = sum(temperatures)
    count = len(temperatures)
    average = total_sum / count
    
    return average
